/*************************************************************** GBoS GEOAI — GAMBIA LULC ATLAS Period: 2000–2025 REGION → DISTRICT → WARD SELECTED-AREA-ONLY LULC PROCESSING Administrative asset: projects/ee-aishaansu80/assets/GambiaEA_Final1 Reference data: ESA WorldCover 2021 v200 Satellite data: Landsat Collection 2 Level 2 Classification: Random Forest — 6 LULC classes Produced by Ansu Singhateh GIS Analyst | GIS / Cartography Unit Gambia Bureau of Statistics (GBoS) GBoS GEOAI • Research and educational use ***************************************************************/ // ============================================================= // 1. CONFIGURATION // ============================================================= var ADMIN_ASSET = 'projects/ee-aishaansu80/assets/GambiaEA_Final1'; var START_YEAR = 2000; var END_YEAR = 2025; var REFERENCE_YEAR = 2021; var CLASSIFICATION_SCALE = 30; var STATISTICS_SCALE = 120; var TRAINING_POINTS_PER_CLASS = 120; var PREDICTOR_BANDS = [ 'blue', 'green', 'red', 'nir', 'swir1', 'swir2', 'NDVI', 'NDWI', 'NDBI' ]; var CLASS_NAMES = [ 'Forest', 'Other vegetation', 'Cropland', 'Built-up', 'Bare land', 'Water' ]; var CLASS_PALETTE = [ '145A32', '7DCEA0', 'F4D03F', 'D73027', 'C2B280', '2471A3' ]; var WORLD_COVER_VALUES = [ 10, 20, 30, 40, 50, 60, 80, 90, 95, 100 ]; var WORLD_COVER_CLASSES = [ 0, 1, 1, 2, 3, 4, 5, 1, 0, 1 ]; // ============================================================= // 2. ADMINISTRATIVE BOUNDARIES // ============================================================= var adminRaw = ee.FeatureCollection(ADMIN_ASSET); // Standardise administrative fields into text keys. // This also accommodates numeric region codes. var admin = adminRaw.map(function(feature) { var region = ee.String( ee.Algorithms.If( feature.get('RCO'), feature.get('RCO'), 'Unknown' ) ); var district = ee.String( ee.Algorithms.If( feature.get('distname'), feature.get('distname'), 'Unknown' ) ); var ward = ee.String( ee.Algorithms.If( feature.get('wardname'), feature.get('wardname'), 'Unknown' ) ); return feature.set({ region_key: region, district_key: district, ward_key: ward }); }); print('Administrative EA features:', admin.size()); print( 'Administrative fields:', admin.first() ); // ============================================================= // 3. ESA WORLDCOVER REFERENCE // ============================================================= var worldCover = ee.ImageCollection( 'ESA/WorldCover/v200' ).first().select('Map'); var referenceLULC = worldCover .remap( WORLD_COVER_VALUES, WORLD_COVER_CLASSES ) .rename('lc') .toInt(); // ============================================================= // 4. LANDSAT PRE-PROCESSING // ============================================================= function maskLandsat(image) { var qa = image.select('QA_PIXEL'); var mask = qa.bitwiseAnd(1 << 0).eq(0) .and(qa.bitwiseAnd(1 << 1).eq(0)) .and(qa.bitwiseAnd(1 << 2).eq(0)) .and(qa.bitwiseAnd(1 << 3).eq(0)) .and(qa.bitwiseAnd(1 << 4).eq(0)) .and(qa.bitwiseAnd(1 << 5).eq(0)); var saturationMask = image .select('QA_RADSAT') .eq(0); return image .updateMask(mask) .updateMask(saturationMask); } // Landsat 5 and Landsat 7 function prepareL57(image) { image = maskLandsat(image); var optical = image.select( [ 'SR_B1', 'SR_B2', 'SR_B3', 'SR_B4', 'SR_B5', 'SR_B7' ], [ 'blue', 'green', 'red', 'nir', 'swir1', 'swir2' ] ); optical = optical .multiply(0.0000275) .add(-0.2); return addIndices(optical) .copyProperties(image, ['system:time_start']); } // Landsat 8 and Landsat 9 function prepareL89(image) { image = maskLandsat(image); var optical = image.select( [ 'SR_B2', 'SR_B3', 'SR_B4', 'SR_B5', 'SR_B6', 'SR_B7' ], [ 'blue', 'green', 'red', 'nir', 'swir1', 'swir2' ] ); optical = optical .multiply(0.0000275) .add(-0.2); return addIndices(optical) .copyProperties(image, ['system:time_start']); } // Add spectral indices function addIndices(image) { var ndvi = image.normalizedDifference( ['nir', 'red'] ).rename('NDVI'); var ndwi = image.normalizedDifference( ['green', 'nir'] ).rename('NDWI'); var ndbi = image.normalizedDifference( ['swir1', 'nir'] ).rename('NDBI'); return image.addBands([ ndvi, ndwi, ndbi ]); } // ============================================================= // 5. ANNUAL LANDSAT COMPOSITE // ============================================================= function getAnnualComposite(year, aoi) { var start = ee.Date.fromYMD(year, 1, 1); var end = start.advance(1, 'year'); var l5 = ee.ImageCollection( 'LANDSAT/LT05/C02/T1_L2' ) .filterBounds(aoi) .filterDate(start, end) .map(prepareL57); var l7 = ee.ImageCollection( 'LANDSAT/LE07/C02/T1_L2' ) .filterBounds(aoi) .filterDate(start, end) .map(prepareL57); var l8 = ee.ImageCollection( 'LANDSAT/LC08/C02/T1_L2' ) .filterBounds(aoi) .filterDate(start, end) .map(prepareL89); var l9 = ee.ImageCollection( 'LANDSAT/LC09/C02/T1_L2' ) .filterBounds(aoi) .filterDate(start, end) .map(prepareL89); var collection = l5 .merge(l7) .merge(l8) .merge(l9); return { collection: collection, image: collection .median() .select(PREDICTOR_BANDS) .clip(aoi) }; } // ============================================================= // 6. SELECTED AREA FILTERING // ============================================================= function getSelectedArea() { var selectedRegion = regionSelect.getValue(); var selectedDistrict = districtSelect.getValue(); var selectedWard = wardSelect.getValue(); var selected = admin; if (selectedRegion !== 'All regions') { selected = selected.filter( ee.Filter.eq( 'region_key', selectedRegion ) ); } if (selectedDistrict !== 'All districts') { selected = selected.filter( ee.Filter.eq( 'district_key', selectedDistrict ) ); } if (selectedWard !== 'All wards') { selected = selected.filter( ee.Filter.eq( 'ward_key', selectedWard ) ); } return selected; } // ============================================================= // 7. SELECTED AREA GEOMETRY // ============================================================= function getAOI(selectedFeatures) { return selectedFeatures .geometry(1000) .simplify(500); } // ============================================================= // 8. TRAINING DATA // ============================================================= function createTrainingData(aoi, referenceComposite) { // Both the reference labels and predictor image are // restricted to the selected AOI. var reference = referenceLULC .clip(aoi); var trainingStack = referenceComposite .select(PREDICTOR_BANDS) .addBands(reference); var samples = trainingStack.stratifiedSample({ numPoints: TRAINING_POINTS_PER_CLASS, classBand: 'lc', region: aoi, scale: CLASSIFICATION_SCALE, seed: 42, geometries: false, dropNulls: true, tileScale: 8 }); return samples; } // ============================================================= // 9. RANDOM FOREST CLASSIFICATION // ============================================================= function classifyImage(image, trainingSamples) { var classifier = ee.Classifier.smileRandomForest({ numberOfTrees: 100, variablesPerSplit: null, minLeafPopulation: 2, bagFraction: 0.7, seed: 42 }).train({ features: trainingSamples, classProperty: 'lc', inputProperties: PREDICTOR_BANDS }); return image .select(PREDICTOR_BANDS) .classify(classifier) .rename('LULC'); } // ============================================================= // 10. CLASS AREA STATISTICS // ============================================================= function calculateAreaStatistics(classified, aoi) { var pixelArea = ee.Image.pixelArea(); var areaImage = pixelArea .rename('area') .addBands( classified.rename('class') ); var result = areaImage.reduceRegion({ reducer: ee.Reducer.sum().group({ groupField: 1, groupName: 'class' }), geometry: aoi, scale: STATISTICS_SCALE, maxPixels: 1e13, bestEffort: true, tileScale: 8 }); var groups = ee.List( ee.Dictionary(result).get('groups', ee.List([])) ); var classNameList = ee.List(CLASS_NAMES); var statistics = ee.FeatureCollection( groups.map(function(item) { var group = ee.Dictionary(item); var classValue = ee.Number( group.get('class') ).toInt(); var areaHa = ee.Number( group.get('sum') ).divide(10000); return ee.Feature(null, { class_id: classValue, class_name: classNameList.get(classValue), area_ha: areaHa, area_km2: areaHa.divide(100) }); }) ); var totalArea = ee.Number( statistics.aggregate_sum('area_ha') ); statistics = statistics.map(function(feature) { var area = ee.Number(feature.get('area_ha')); var percentage = ee.Number( ee.Algorithms.If( totalArea.gt(0), area.divide(totalArea).multiply(100), 0 ) ); return feature.set({ percentage: percentage }); }); return { collection: statistics, totalArea: totalArea }; } // ============================================================= // 11. CHANGE STATISTICS // ============================================================= function calculateChangeArea(classified, comparison) { var changed = classified .neq(comparison) .rename('change') .selfMask(); var area = ee.Image.pixelArea() .updateMask(changed); var result = area.reduceRegion({ reducer: ee.Reducer.sum(), geometry: currentResults.aoi, scale: STATISTICS_SCALE, maxPixels: 1e13, bestEffort: true, tileScale: 8 }); return ee.Number( result.get('area', 0) ).divide(10000); } // ============================================================= // 12.
